app/updateParser.js

Parsea lo que viene de Telegram:

function parseTelegramUpdate_(e) {
  const contents = JSON.parse(e.postData.contents);

  const text = contents.message?.text?.trim();
  const chatId = String(contents.message?.chat?.id || "");

  if (!text || !chatId) return null;

  return {
    chatId,
    text,
    raw: contents
  };
}
app/commandRouter.js

Decide qué comando ejecutar:

function handleCommand_(update) {
  const parts = parseTelegramMessage_(update.text);
  const commandName = normalizeCommandName_(parts[0]);
  const entry = COMMAND_INDEX[commandName];

  if (!entry) {
    sendTelegram(MSG_ERRORS.INVALID_COMMAND);
    return;
  }

  executeCommand_(update.chatId, parts, entry);
}
app/stateRouter.js

Decide qué handler de estado ejecutar:

function handleState_(update, state) {
  const flowHandlers = STATE_HANDLERS[state.flow];

  if (!flowHandlers) {
    StateService.clear(update.chatId);
    sendTelegram("❌ Estado inválido. Volvé a empezar.");
    return;
  }

  const handler = flowHandlers[state.step];

  if (!handler) {
    StateService.clear(update.chatId);
    sendTelegram("❌ Paso inválido. Volvé a empezar.");
    return;
  }

  handler(update.chatId, update.text, state);
}

expenseValidator.js

Valida y normaliza.

var ExpenseValidator = {
  validate(input) {
    const errors = [];

    const expense = {
      fecha: processDateInput(input.fecha, errors),
      monto: processAmountInput(input.monto, errors),
      moneda: processCoinInput(input.moneda, errors),
      metodo: processMethodInput(input.metodo, errors),
      detalle: processDetailInput(input.detalle, errors),
      tipo: processTypeInput(input.tipo, errors)
    };

    expense.ahorro = processSavingInput(expense.monto, input.ahorro, errors);
    expense.reintegrado = processRefundInput(input.reintegrado, expense.tipo, expense.ahorro, errors);

    if (errors.length > 0) {
      return {
        ok: false,
        errors
      };
    }

    return {
      ok: true,
      value: expense
    };
  }
};

Regla simple para saber dónde va cada función

Usá esta pregunta:

¿La función recibe texto del usuario y lo parte?

Idea central

La regla más importante es esta:

main.js no sabe de funcionalidades.
commandRouter.js no sabe de Sheets.
services no saben de gastos.
repositories no mandan mensajes.
domain no manipula rangos directamente.

Si mantenés eso, agregar funcionalidades nuevas como:

editar gastos
eliminar deudas
pausar recordatorios
inversiones
gastos compartidos

va a ser mucho más fácil.
